/*
===============================================================================
 Jenkinsfile – pyats-config-push
 Version : v2.2.1
-------------------------------------------------------------------------------
 FIX IN THIS VERSION:
 - Use pyats-venv/bin/python explicitly
 - ELIMINATES "ModuleNotFoundError: pyats"
 - Jenkins-safe virtualenv execution
===============================================================================
*/

pipeline {
    agent any

    options {
        timestamps()
        ansiColor('xterm')
    }

    parameters {
        string(
            name: 'TESTBED_FILE',
            defaultValue: 'testbeds/testbed_access_2960.yaml',
            description: 'Path to pyATS testbed YAML file'
        )

        text(
            name: 'CONFIG_COMMANDS',
            defaultValue: '',
            description: 'Multiline IOS CLI configuration to push'
        )

        string(
            name: 'DEVICE_FILTER',
            defaultValue: 'ALL',
            description: 'Comma-separated device list OR ALL'
        )

        booleanParam(
            name: 'WRITE_MEMORY',
            defaultValue: true,
            description: 'Run "write memory" after pushing config'
        )
    }

    stages {

        stage('Checkout SCM') {
            steps {
                checkout scm
            }
        }

        stage('Prepare Virtualenv') {
            steps {
                sh '''
                    set -e
                    if [ ! -d pyats-venv ]; then
                        python3.11 -m venv pyats-venv
                    fi

                    pyats-venv/bin/pip install --upgrade pip
                    pyats-venv/bin/pip install pyats==24.3 genie==24.3 unicon==24.3
                '''
            }
        }

        stage('Verify pyATS Environment') {
            steps {
                sh '''
                    set -e
                    pyats-venv/bin/python - << 'EOF'
import pyats
import genie
import unicon
print("pyATS OK")
EOF
                '''
            }
        }

        stage('Push Config (Parallel per Device)') {
            steps {
                script {

                    def devicesRaw = sh(
                        script: """
                            pyats-venv/bin/python scripts/config_push.py \
                              --testbed ${params.TESTBED_FILE} \
                              --list-devices
                        """,
                        returnStdout: true
                    ).trim()

                    def devices = devicesRaw
                        .split("\\n")
                        .collect { it.trim() }
                        .findAll { it }

                    def branches = [:]

                    for (int i = 0; i < devices.size(); i++) {
                        def deviceName = devices[i]

                        branches[deviceName] = {
                            stage("Push → ${deviceName}") {
                                sh """
                                    set -e
                                    pyats-venv/bin/python scripts/config_push.py \
                                      --testbed ${params.TESTBED_FILE} \
                                      --devices ${deviceName} \
                                      --config "${params.CONFIG_COMMANDS}" \
                                      ${params.WRITE_MEMORY ? '--write-memory' : ''}
                                """
                            }
                        }
                    }

                    parallel branches
                }
            }
        }
    }

    post {
        success {
            echo "✅ All devices completed successfully"
        }
        failure {
            echo "❌ Pipeline failed"
        }
    }
}
